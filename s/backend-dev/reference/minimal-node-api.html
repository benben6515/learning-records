The skeleton, annotated

import { createServer } from "node:http";

// store → becomes a DB in Lesson 05
const notes = new Map();
let nextId = 1;

// body: stream → promise → string (parse in the handler)
function readBody(req) {
  return new Promise((resolve) => {
    let body = "";
    req.on("data", (c) => (body += c));
    req.on("end", () => resolve(body));
  });
}

// response: status + headers + body in one call
function json(res, status, data) {
  res.writeHead(status, { "content-type": "application/json" });
  res.end(data === undefined ? "" : JSON.stringify(data));
}

// route table: { method, pattern, handler } — plain data
const routes = [
  { method: "GET",    pattern: "/notes",     handler: … },
  { method: "POST",   pattern: "/notes",     handler: … },  // 400 on missing title, 201 on create
  { method: "GET",    pattern: "/notes/:id", handler: … },  // 200 or 404
  { method: "DELETE", pattern: "/notes/:id", handler: … },  // 204 or 404
];

// ":id" → named capture group, digits only, optional trailing slash
const toRegex = (p) => new RegExp("^" + p.replace(/:(\w+)/g, "(?<$1>\\d+)") + "/?$");

// dispatch: match → run (await feeds rejections to the catch!) → 500
const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, "http://localhost");
  try {
    for (const route of routes) {
      if (req.method !== route.method) continue;
      const match = pathname.match(toRegex(route.pattern));
      if (match) return await route.handler(req, res, match.groups ?? {});
    }
    json(res, 404, { error: "not found" });
  } catch (err) {
    console.error(err);
    json(res, 500, { error: "internal error" });
  }
});

server.listen(3000, () => console.log("listening on http://localhost:3000"));
