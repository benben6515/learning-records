骨架，附註解

import { createServer } from "node:http";

// 資料存放 → 第 05 課換成資料庫
const notes = new Map();
let nextId = 1;

// 內文：串流 → promise → 字串（在 handler 裡 parse）
function readBody(req) {
  return new Promise((resolve) => {
    let body = "";
    req.on("data", (c) => (body += c));
    req.on("end", () => resolve(body));
  });
}

// 回應：狀態＋標頭＋內文，一次呼叫
function json(res, status, data) {
  res.writeHead(status, { "content-type": "application/json" });
  res.end(data === undefined ? "" : JSON.stringify(data));
}

// 路由表：{ method, pattern, handler }——普通資料
const routes = [
  { method: "GET",    pattern: "/notes",     handler: … },
  { method: "POST",   pattern: "/notes",     handler: … },  // 缺 title 回 400，建立成功 201
  { method: "GET",    pattern: "/notes/:id", handler: … },  // 200 或 404
  { method: "DELETE", pattern: "/notes/:id", handler: … },  // 204 或 404
];

// ":id" → 具名擷取群組、僅數字、容許結尾斜線
const toRegex = (p) => new RegExp("^" + p.replace(/:(\w+)/g, "(?<$1>\\d+)") + "/?$");

// 分派：配對 → 執行（await 才能把 rejection 餵進 catch！）→ 500
const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, "http://localhost");
  try {
    for (const route of routes) {
      if (req.method !== route.method) continue;
      const match = pathname.match(toRegex(route.pattern));
      if (match) return await route.handler(req, res, match.groups ?? {});
    }
    json(res, 404, { error: "not found" });
  } catch (err) {
    console.error(err);
    json(res, 500, { error: "internal error" });
  }
});

server.listen(3000, () => console.log("listening on http://localhost:3000"));
